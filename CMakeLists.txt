cmake_minimum_required(VERSION 3.20)
project(TinySolverDx VERSION 0.1.0 LANGUAGES CXX)

option(TSDX_BUILD_TESTS "Build GoogleTest suite" ON)
option(TSDX_BUILD_BENCH "Build Google Benchmark micro-benchmarks" ON)
option(TSDX_BUILD_PYTHON "Build the Python extension (_core)" ON)
option(TSDX_ENABLE_CUDA "Build the CUDA backend if nvcc is available" ON)
option(TSDX_NATIVE "Compile with -march=native / -mcpu=native when supported" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(CheckCXXCompilerFlag)
include(CheckLanguage)

add_library(tsdx STATIC src/runtime.cpp)
target_include_directories(tsdx PUBLIC include)
target_compile_features(tsdx PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(tsdx PUBLIC Threads::Threads)

if(TSDX_NATIVE)
  check_cxx_compiler_flag(-march=native TSDX_HAS_MARCH_NATIVE)
  if(TSDX_HAS_MARCH_NATIVE)
    target_compile_options(tsdx PUBLIC -march=native)
  else()
    check_cxx_compiler_flag(-mcpu=native TSDX_HAS_MCPU_NATIVE)
    if(TSDX_HAS_MCPU_NATIVE)
      target_compile_options(tsdx PUBLIC -mcpu=native)
    endif()
  endif()
endif()

find_package(OpenMP COMPONENTS CXX QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(tsdx PUBLIC OpenMP::OpenMP_CXX)
  target_compile_definitions(tsdx PUBLIC TSDX_HAVE_OPENMP=1)
  message(STATUS "TinySolverDx: OpenMP enabled")
else()
  message(STATUS "TinySolverDx: OpenMP not found, using std::thread fallback")
endif()

if(TSDX_ENABLE_CUDA)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES native)
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    target_sources(tsdx PRIVATE src/cuda/cuda_backend.cu)
    target_compile_definitions(tsdx PUBLIC TSDX_WITH_CUDA=1)
    set_target_properties(tsdx PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
    find_package(CUDAToolkit REQUIRED)
    target_link_libraries(tsdx PUBLIC CUDA::cudart_static)
    message(STATUS "TinySolverDx: CUDA backend enabled (${CMAKE_CUDA_COMPILER})")
  else()
    message(STATUS "TinySolverDx: nvcc not found, building CPU backends only")
  endif()
endif()

include(FetchContent)

if(TSDX_BUILD_TESTS)
  enable_testing()
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  add_executable(tsdx_tests tests/cpp/test_precision.cpp tests/cpp/test_solvers.cpp)
  target_link_libraries(tsdx_tests PRIVATE tsdx GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(tsdx_tests DISCOVERY_TIMEOUT 120)
endif()

if(TSDX_BUILD_BENCH)
  find_package(benchmark QUIET)
  if(NOT benchmark_FOUND)
    FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.1 GIT_SHALLOW TRUE)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_GTEST_TESTS OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(benchmark)
    # newer clang flags benchmark's __COUNTER__ use under its -pedantic-errors
    target_compile_options(benchmark PRIVATE -Wno-c2y-extensions)
    target_compile_options(benchmark INTERFACE -Wno-c2y-extensions)
  endif()
  add_executable(tsdx_bench bench/bench_cpu.cpp)
  target_link_libraries(tsdx_bench PRIVATE tsdx benchmark::benchmark)
endif()

if(TSDX_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir
                  OUTPUT_VARIABLE TSDX_PYBIND11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TSDX_PB_RC ERROR_QUIET)
  if(TSDX_PB_RC EQUAL 0)
    list(APPEND CMAKE_PREFIX_PATH "${TSDX_PYBIND11_DIR}")
    find_package(pybind11 CONFIG REQUIRED)
  else()
    FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v2.13.6 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(pybind11)
  endif()
  pybind11_add_module(_core src/python/bindings.cpp)
  target_link_libraries(_core PRIVATE tsdx)
  set_target_properties(_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/python/tinysolverdx)
endif()
